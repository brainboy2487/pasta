# Validate vocabulary JSONL files
.PHONY: validate-vocab ci
validate-vocab:
	@if [ -z "$(VOCAB_FILES)" ]; then \
		echo "No vocab files found (glob: data/vocab_*.jsonl)"; exit 0; \
	fi
	@for f in $(VOCAB_FILES); do \
		echo "Validating $$f"; \
		$(VALIDATE_PY) $$f || exit 1; \
	done

# Validate the canonical generated vocab file
.PHONY: validate-generated-vocab
validate-generated-vocab:
	@echo "Validating data/vocab/generated_vocab.jsonl"
	@./$(BIN_DIR)/mage_cli validate-vocab data/vocab/generated_vocab.jsonl || exit 1
